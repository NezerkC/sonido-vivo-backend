# Despliegue e Inicialización: Oracle 23ai y MongoDB en Proxmox CT100

**Proyecto:** Sonido Vivo Backend (`sonido-vivo-backend`)  
**Infraestructura:** Host Proxmox `192.168.3.3` (pvehplaptop) | Contenedor Docker Engine `192.168.3.150` (CT100)  
**Objetivo:** Resolver el error OOM 137 del contenedor de Oracle Database 23ai Free, desplegar la base NoSQL MongoDB (`sonido_vivo_nosql`), compilar el paquete PL/SQL `PKG_SONIDO_VIVO` y parametrizar la conectividad del backend hexagonal.

---

## 1. Diagnóstico de Capacidad y Causa Raíz OOM 137

### 1.1 Presupuesto de Memoria RAM
* **Host Hipervisor (Proxmox VE 8):** 7.7 GiB físicos totales (~3.8 GiB utilizados por servicios base y reserva del host).
* **Contenedor Principal (CT100):** Límite configurado de **6.5 GiB RAM**.
  * Consumo actual base: ~1.9 GiB (Dokploy, Traefik, `reactsonidovivo-sekgas`).
  * Memoria libre disponible en CT100: **~4.6 GiB**.

### 1.2 Causa Raíz de `OOMKilled=true` (Exit Code 137)
El archivo `/opt/oracle-hermes/docker-compose.yml` en CT100 tenía configurado un límite estricto de memoria de `2048M` (`mem_limit: 2g`).  
El motor **Oracle Database 23ai Free** requiere un mínimo de **2.5 GiB a 3.2 GiB** durante la fase de inicialización de la instancia (SGA + PGA compartida) y el montaje del PDB (`FREEPDB1`). Al alcanzar los 2048 MB, el OOM-killer de Linux envió la señal `SIGKILL` (código de salida `137`).

## 3. Inicialización Relacional: `PKG_SONIDO_VIVO` (Oracle PL/SQL)

Ruta del script en CT100: `/opt/oracle-hermes/scripts/oracle/01_pkg_sonido_vivo.sql`

```sql
ALTER SESSION SET CONTAINER = FREEPDB1;

-- Crear esquema de aplicación
CREATE USER C##SONIDO_VIVO IDENTIFIED BY "SonidoVivo2026#Secure";
GRANT CONNECT, RESOURCE, DBA TO C##SONIDO_VIVO;
ALTER USER C##SONIDO_VIVO QUOTA UNLIMITED ON USERS;

CONNECT C##SONIDO_VIVO/"SonidoVivo2026#Secure"@//localhost:1521/FREEPDB1;

-- Tablas relacionales transaccionales
CREATE TABLE clientes (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    rut VARCHAR2(12) NOT NULL UNIQUE,
    nombre VARCHAR2(100) NOT NULL,
    email VARCHAR2(150) NOT NULL UNIQUE,
    tipo VARCHAR2(30) DEFAULT 'ESTANDAR' CHECK (tipo IN ('ESTANDAR', 'FRECUENTE', 'PREMIUM')),
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ventas (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    numero_boleta VARCHAR2(50) NOT NULL UNIQUE,
    rut_cliente VARCHAR2(12) NOT NULL,
    monto_total_clp NUMBER(12, 0) NOT NULL,
    descuento_clp NUMBER(12, 0) DEFAULT 0,
    monto_final_clp NUMBER(12, 0) NOT NULL,
    estado VARCHAR2(20) DEFAULT 'COMPLETADA',
    fecha_venta TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_ventas_cliente FOREIGN KEY (rut_cliente) REFERENCES clientes(rut)
);

CREATE TABLE detalle_ventas (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    venta_id NUMBER NOT NULL,
    codigo_producto VARCHAR2(50) NOT NULL,
    cantidad NUMBER(6, 0) NOT NULL,
    precio_unitario_clp NUMBER(12, 0) NOT NULL,
    subtotal_clp NUMBER(12, 0) NOT NULL,
    CONSTRAINT fk_detalle_venta FOREIGN KEY (venta_id) REFERENCES ventas(id)
);

-- Cabecera del Paquete
CREATE OR REPLACE PACKAGE PKG_SONIDO_VIVO AS
    PROCEDURE SP_REGISTRAR_VENTA(
        p_rut_cliente       IN VARCHAR2,
        p_monto_total       IN NUMBER,
        p_codigo_producto   IN VARCHAR2,
        p_cantidad          IN NUMBER,
        p_precio_unitario   IN NUMBER,
        p_boleta_generada   OUT VARCHAR2,
        p_monto_final       OUT NUMBER
    );

    PROCEDURE SP_LIQUIDAR_CLIENTES(
        p_rut_cliente       IN VARCHAR2,
        p_total_ventas      OUT NUMBER,
        p_ventas_count      OUT NUMBER
    );

    FUNCTION FN_CALCULAR_DESCUENTO(
        p_rut_cliente       IN VARCHAR2,
        p_monto_base        IN NUMBER
    ) RETURN NUMBER;
END PKG_SONIDO_VIVO;
/

-- Cuerpo del Paquete
CREATE OR REPLACE PACKAGE BODY PKG_SONIDO_VIVO AS

    FUNCTION FN_CALCULAR_DESCUENTO(
        p_rut_cliente IN VARCHAR2,
        p_monto_base  IN NUMBER
    ) RETURN NUMBER IS
        v_tipo VARCHAR2(30);
        v_descuento NUMBER := 0;
    BEGIN
        SELECT NVL(tipo, 'ESTANDAR') INTO v_tipo FROM clientes WHERE rut = p_rut_cliente;
        IF v_tipo = 'PREMIUM' THEN
            v_descuento := ROUND(p_monto_base * 0.15);
        ELSIF v_tipo = 'FRECUENTE' THEN
            v_descuento := ROUND(p_monto_base * 0.08);
        END IF;
        RETURN v_descuento;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RETURN 0;
    END FN_CALCULAR_DESCUENTO;

    PROCEDURE SP_REGISTRAR_VENTA(
        p_rut_cliente       IN VARCHAR2,
        p_monto_total       IN NUMBER,
        p_codigo_producto   IN VARCHAR2,
        p_cantidad          IN NUMBER,
        p_precio_unitario   IN NUMBER,
        p_boleta_generada   OUT VARCHAR2,
        p_monto_final       OUT NUMBER
    ) IS
        v_descuento NUMBER := 0;
        v_venta_id  NUMBER;
    BEGIN
        v_descuento := FN_CALCULAR_DESCUENTO(p_rut_cliente, p_monto_total);
        p_monto_final := p_monto_total - v_descuento;
        p_boleta_generada := 'BOL-' || TO_CHAR(SYSTIMESTAMP, 'YYYYMMDDHH24MISS') || '-' || DBMS_RANDOM.STRING('X', 4);

        INSERT INTO ventas (numero_boleta, rut_cliente, monto_total_clp, descuento_clp, monto_final_clp)
        VALUES (p_boleta_generada, p_rut_cliente, p_monto_total, v_descuento, p_monto_final)
        RETURNING id INTO v_venta_id;

        INSERT INTO detalle_ventas (venta_id, codigo_producto, cantidad, precio_unitario_clp, subtotal_clp)
        VALUES (v_venta_id, p_codigo_producto, p_cantidad, p_precio_unitario, p_cantidad * p_precio_unitario);

        COMMIT;
    END SP_REGISTRAR_VENTA;

    PROCEDURE SP_LIQUIDAR_CLIENTES(
        p_rut_cliente       IN VARCHAR2,
        p_total_ventas      OUT NUMBER,
        p_ventas_count      OUT NUMBER
    ) IS
    BEGIN
        SELECT NVL(SUM(monto_final_clp), 0), COUNT(*)
        INTO p_total_ventas, p_ventas_count
        FROM ventas
        WHERE rut_cliente = p_rut_cliente;
    END SP_LIQUIDAR_CLIENTES;

END PKG_SONIDO_VIVO;
/
```

## 6. Procedimiento de Ejecución en CT100

1. **Conectar por SSH a CT100:**
   ```bash
   ssh root@192.168.3.150
   ```
2. **Crear directorios de scripts y datos:**
   ```bash
   mkdir -p /opt/oracle-hermes/scripts/oracle /opt/oracle-hermes/scripts/mongo
   mkdir -p /opt/oracle-hermes/oradata /opt/oracle-hermes/mongodata
   ```
3. **Copiar los scripts de inicialización y levantar el stack:**
   ```bash
   cd /opt/oracle-hermes
   docker compose down
   docker compose up -d
   ```
4. **Verificar estado de salud y logs sin OOM:**
   ```bash
   docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"
   docker stats --no-stream
   ```
5. **Verificar inicialización de Oracle:**
   ```bash
   docker logs -f oracle-23ai-sonidovivo
   # Esperar mensaje: "DATABASE IS READY TO USE!"
   ```
6. **Verificar `/actuator/health` en el Backend:**
   ```bash
   curl -s http://localhost:8081/actuator/health | jq .
   ```
